    <script>
        // Interactive Calculator Logic
        const salesRange = document.getElementById('salesRange');
        const salesValue = document.getElementById('salesValue');
        const calcRevenue = document.getElementById('calcRevenue');
        const calcProfit = document.getElementById('calcProfit');

        function updateCalculator() {
            const qty = parseInt(salesRange.value);
            const price = 15;
            const revenue = qty * price;
            const profit = revenue; // 0 variable cost

            salesValue.textContent = `${qty} ${qty === 1 ? 'Pack' : 'Packs'}`;
            calcRevenue.innerHTML = `${revenue} <span class="text-sm">AED</span>`;
            calcProfit.innerHTML = `${profit} <span class="text-xl">AED</span>`;
        }

        salesRange.addEventListener('input', updateCalculator);

        // Modal Functions
        function openModal(type) {
            const overlay = document.getElementById('modalOverlay');
            const buyContent = document.getElementById('modalContentBuy');
            const successContent = document.getElementById('modalContentSuccess');

            buyContent.classList.remove('hidden');
            successContent.classList.add('hidden');
            overlay.classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('modalOverlay').classList.add('hidden');
        }

        function handleDownload(e) {
            e.preventDefault();
            document.getElementById('modalContentBuy').classList.add('hidden');
            document.getElementById('modalContentSuccess').classList.remove('hidden');
        }
